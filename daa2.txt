import heapq

class Node:
    def __init__(self, ch, freq):
        self.ch, self.freq = ch, freq
        self.left = self.right = None

def generate_codes(root, code, result):
    if root is None:
        return
    if root.left is None and root.right is None:             # leaf node
        result.append(f"{root.ch}: {code}")
        return
    generate_codes(root.left, code + "0", result)
    generate_codes(root.right, code + "1", result)

def huffman_encode(chars, freqs):
    # heap item = (freq, creation order, node): ties broken by order, same as Java
    heap = [(f, i, Node(c, f)) for i, (c, f) in enumerate(zip(chars, freqs))]
    heapq.heapify(heap)
    order = len(heap)

    while len(heap) > 1:                                     # greedy: merge 2 smallest
        f1, _, left = heapq.heappop(heap)
        f2, _, right = heapq.heappop(heap)
        new_node = Node('-', f1 + f2)
        new_node.left, new_node.right = left, right
        heapq.heappush(heap, (new_node.freq, order, new_node))
        order += 1

    result = []
    generate_codes(heap[0][2], "", result)
    print("\n".join(result))

chars = ['p', 'q', 'r', 's', 't', 'u']
freqs = [5, 7, 10, 15, 20, 35]
print("Huffman Codes:")
huffman_encode(chars, freqs)

'''
OUTPUT:
Huffman Codes:
s: 00
t: 01
r: 100
p: 1010
q: 1011
u: 11
'''































/*

Title: Huffman Encoding Using Greedy Strategy

CO4, CO5

Program: Write a program to implement Huffman Encoding using a greedy strategy.
===========================================================

import java.util.PriorityQueue;
import java.util.Comparator;

public class Assign2 {
    static int counter = 0;

    static class Node {
        char ch; int freq; int order; Node left, right;
        Node(char ch, int freq) {
            this.ch = ch; this.freq = freq; this.order = counter++;
            this.left = null; this.right = null;
        }
    }

    // lowest frequency first; ties broken by creation order so output is always the same
    static class NodeComparator implements Comparator<Node> {
        public int compare(Node a, Node b) {
            return a.freq != b.freq ? a.freq - b.freq : a.order - b.order;
        }
    }

    public static void generateCodes(Node root, String code, StringBuilder result) {
        if (root == null) return;
        if (root.left == null && root.right == null) {          // leaf node
            result.append(root.ch).append(": ").append(code).append("\n");
            return;
        }
        generateCodes(root.left, code + "0", result);
        generateCodes(root.right, code + "1", result);
    }

    public static void huffmanEncode(char[] chars, int[] freqs) {
        PriorityQueue<Node> pq = new PriorityQueue<>(new NodeComparator());
        for (int i = 0; i < chars.length; i++) pq.add(new Node(chars[i], freqs[i]));

        while (pq.size() > 1) {                                  // greedy: merge 2 smallest
            Node left = pq.poll();
            Node right = pq.poll();
            Node newNode = new Node('-', left.freq + right.freq);
            newNode.left = left;
            newNode.right = right;
            pq.add(newNode);
        }

        StringBuilder result = new StringBuilder();
        generateCodes(pq.poll(), "", result);
        System.out.print(result);
    }

    public static void main(String[] args) {
        char[] chars = {'p', 'q', 'r', 's', 't', 'u'};
        int[] freqs  = {5, 7, 10, 15, 20, 35};
        System.out.println("Huffman Codes:");
        huffmanEncode(chars, freqs);
    }
}
