# Importing libraries
import pandas as pd
import numpy as np
import seaborn as sns
import matplotlib.pyplot as plt

# Importing clustering and dimensionality reduction libraries
from sklearn.cluster import KMeans, k_means
from sklearn.decomposition import PCA


# Loading the dataset
df = pd.read_csv("sales_data_sample.csv", sep=",", encoding='Latin-1')

df.head()

# Checking dataset information
df.shape

df.describe()

df.info()

df.isnull().sum()

df.dtypes


# Dropping unnecessary categorical columns and columns having null values
df_drop = [
    'ADDRESSLINE1', 'ADDRESSLINE2', 'STATUS', 'POSTALCODE',
    'CITY', 'TERRITORY', 'PHONE', 'STATE',
    'CONTACTFIRSTNAME', 'CONTACTLASTNAME',
    'CUSTOMERNAME', 'ORDERNUMBER'
]

df = df.drop(df_drop, axis=1)

df.isnull().sum()

df.dtypes


# Checking categorical columns
df['COUNTRY'].unique()

df['PRODUCTLINE'].unique()

df['DEALSIZE'].unique()


# Converting categorical columns into numerical columns
productline = pd.get_dummies(df['PRODUCTLINE'], dtype=int)

Dealsize = pd.get_dummies(df['DEALSIZE'], dtype=int)

df = pd.concat([df, productline, Dealsize], axis=1)


# Dropping original categorical columns
df_drop = ['COUNTRY', 'PRODUCTLINE', 'DEALSIZE']

df = df.drop(df_drop, axis=1)


# Converting PRODUCTCODE into numerical values
df['PRODUCTCODE'] = pd.Categorical(df['PRODUCTCODE']).codes


# Dropping ORDERDATE column
df.drop('ORDERDATE', axis=1, inplace=True)


# Checking all datatypes
df.dtypes


# ELBOW METHOD

distortions = []

# Testing number of clusters from 1 to 9
K = range(1, 10)

for k in K:

    kmeanModel = KMeans(n_clusters=k, n_init=10)

    kmeanModel.fit(df)

    distortions.append(kmeanModel.inertia_)


# Plotting the elbow graph
plt.figure(figsize=(16, 8))

plt.plot(K, distortions, 'bx-')

plt.xlabel('k')

plt.ylabel('Distortion')

plt.title('The Elbow Method showing the optimal k')

plt.show()


# Converting dataframe into NumPy array
X_train = df.values

X_train.shape


# Creating K-Means model with 3 clusters
model = KMeans(n_clusters=3, random_state=2, n_init=10)

# Fitting the model
model = model.fit(X_train)


# Predicting cluster values
predictions = model.predict(X_train)


# Counting data points in each cluster
unique, counts = np.unique(predictions, return_counts=True)

counts = counts.reshape(1, 3)

counts_df = pd.DataFrame(
    counts,
    columns=['Cluster1', 'Cluster2', 'Cluster3']
)

counts_df.head()


# Applying PCA for dimensionality reduction
pca = PCA(n_components=2)

# Converting all features into 2 components
reduced_X = pd.DataFrame(
    pca.fit_transform(X_train),
    columns=['PCA1', 'PCA2']
)

reduced_X.head()


# Plotting normal scatter plot
plt.figure(figsize=(14, 10))

plt.scatter(reduced_X['PCA1'], reduced_X['PCA2'])


# Finding cluster centroids
model.cluster_centers_


# Transforming centroids into PCA coordinates
reduced_centers = pca.transform(model.cluster_centers_)

reduced_centers


# Plotting scatter plot with centroids
plt.figure(figsize=(14, 10))

plt.scatter(reduced_X['PCA1'], reduced_X['PCA2'])

plt.scatter(
    reduced_centers[:, 0],
    reduced_centers[:, 1],
    color='black',
    marker='x',
    s=300
)


# Adding cluster predictions to dataframe
reduced_X['Clusters'] = predictions

reduced_X.head()


# Plotting the clusters
plt.figure(figsize=(14, 10))

# Cluster 0
plt.scatter(
    reduced_X[reduced_X['Clusters'] == 0].loc[:, 'PCA1'],
    reduced_X[reduced_X['Clusters'] == 0].loc[:, 'PCA2'],
    color='slateblue'
)

# Cluster 1
plt.scatter(
    reduced_X[reduced_X['Clusters'] == 1].loc[:, 'PCA1'],
    reduced_X[reduced_X['Clusters'] == 1].loc[:, 'PCA2'],
    color='springgreen'
)

# Cluster 2
plt.scatter(
    reduced_X[reduced_X['Clusters'] == 2].loc[:, 'PCA1'],
    reduced_X[reduced_X['Clusters'] == 2].loc[:, 'PCA2'],
    color='indigo'
)

# Plotting centroids
plt.scatter(
    reduced_centers[:, 0],
    reduced_centers[:, 1],
    color='black',
    marker='x',
    s=300
)

plt.xlabel('PCA1')
plt.ylabel('PCA2')

plt.title('Sales Data Clustering using K-Means and PCA')

plt.show()

3. Understand the important concepts
A. What is clustering?
Clustering is an unsupervised machine learning technique that groups similar data points together.

For example, a company can group customers based on their purchasing behavior.








medium.com


B. What is K-Means?
K-Means is an unsupervised machine learning algorithm that divides data into K clusters based on similarity.

Working of K-Means:
1. Select the number of clusters (K).
2. Initialize K centroids.
3. Calculate the distance between data points and centroids.
4. Assign each point to its nearest centroid.
5. Recalculate the centroids.
6. Repeat until clusters stop changing significantly.

- The elbow point indicates a suitable K value.

D. What is PCA?
PCA stands for Principal Component Analysis.

It is a dimensionality reduction technique that reduces the number of features while retaining as much variation in the data as possible.

Your code uses:

This converts all input features into two components:
- PCA1
- PCA2
This makes it easier to visualize the clusters in a 2D graph.
E. What is a centroid?
A centroid is the center point of a cluster.

In your graph, black X marks represent the three cluster centroids.
